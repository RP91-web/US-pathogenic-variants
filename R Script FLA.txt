# Define the function to perform Mann-Whitney test and p-value adjustment
perform_mann_whitney <- function(df, dependent_vars, group_var, keep_vars) {
  # Perform Mann-Whitney tests for each dependent variable
  p_values <- sapply(dependent_vars, function(var) {
    tryCatch({
      wilcox.test(df[[var]] ~ df[[group_var]])$p.value
    }, error = function(e) {
      NA  # Return NA if an error occurs for any variable
    })
  })
  
  # Apply Benjamini-Hochberg correction
  p_adj <- p.adjust(p_values, method = "BH")
  
  # Create a results data frame
  results <- data.frame(Variable = dependent_vars, P_Value = p_values, P_Adj = p_adj)
  
  # Get names of significant variables (p_adj < 0.051)
  significant_vars <- results %>% filter(P_Adj < 0.051) %>% pull(Variable)
  
  # Filter the data to keep only the significant variables
  df_filtered <- df %>% select(all_of(keep_vars), all_of(significant_vars))
  
  # Return filtered dataframe and significant variables
  return(list(filtered_data = df_filtered, significant_vars = significant_vars))
}

# Define the function to perform the analysis on each dataset
perform_analysis <- function(df, start_col, end_col, keep_vars) {
  # Select only the numeric variables from a specific range (columns 14 to 123 in this case)
  numeric_vars_range <- df[, start_col:end_col]
  
  # Remove rows with missing values in the numeric variables range
  numeric_vars_range <- na.omit(numeric_vars_range)
  
  # Create a correlation matrix for the selected numeric variables
  cor_matrix <- cor(numeric_vars_range, method = "spearman")
  
  # View the correlation matrix
  View(cor_matrix)
  
  # Plot the correlation matrix using corrplot
  corrplot(cor_matrix, tl.cex = 0.5)
  
  # Find redundant variables with high correlations (correlation > 0.90)
  redundant_vars <- findCorrelation(cor_matrix, cutoff = 0.90, names = TRUE)
  
  # Display redundant variables
  print(redundant_vars)
  
  # Remove the redundant variables from the selected numeric variables
  selected_vars <- setdiff(names(numeric_vars_range), redundant_vars)
  
  # Create a new dataset with the selected features (including the non-numeric variables)
  new_dataset <- df %>% select(all_of(keep_vars), all_of(selected_vars))
  
  # Return the new dataset with selected features
  return(new_dataset)
}

#Split Data into Training (75%) and Testing (25%)
#Formula derived from:
#Joseph, V. R. (2022). Optimal ratio for data splitting. Statistical Analysis and Data Mining, 15(4), 531–538. doi:10.1002/sam.11583
# Recommended ratio is sqrt(p):1, where p = no. of free parameters
#We used p = sqrt(N)n where N is the number of unique rows in the input matrix of the dataset
#So, for tumorCorr this would be sqrt(86) = 9.27. 
#Thus, the splitting ratio is sqrt(9.27):1 = ~3:1 (75% and 25%)
#For tumorPeriCorr this would be sqrt(88) = 9.38
#Thus, the splitting ratio is sqrt(9.38):1 = ~3:1 (75% and 25%)
###################################################################################################

# Function to split data into 75% training and 25% testing
split_data <- function(data) {
  train_index <- createDataPartition(data$Mutatie_patogena_prezenta, p = 0.75, list = FALSE)
  train_set <- data[train_index, ]
  test_set <- data[-train_index, ]
  return(list(train = train_set, test = test_set))
}
